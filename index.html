📄 ESP32_SolarPowerbank_Telemetry.ino
/* =========================================================================
 * PROYEK: SOLAR POWERBANK TELEMETRY & IoT MQTT HUB
 * Author: M KHOIRUL FAHMI (KELAS XI C - TEKNIK INFORMATIKA)
 * Target Board: ESP32 Mini / D1 Mini ESP32 / DevKit V1
 * Sensor: INA219 (I2C Current Sensor) + Voltage Divider (ADC GPIO34)
 * Cloud: MQTT Broker (Adafruit IO / HiveMQ / Blynk IoT)
 * ========================================================================= */

#include <WiFi.h>
#include <PubSubClient.h>
#include <Wire.h>
#include <Adafruit_INA219.h>

const char* ssid = "Powerbank_IoT_Hotspot";
const char* password = "fahmi12345";
const char* mqtt_server = "broker.hivemq.com";
const char* mqtt_topic = "fahmi/iot/solar/telemetry";

#define PIN_SOLAR_ADC    34
#define PIN_BATTERY_ADC  35
#define PIN_STATUS_LED   2

Adafruit_INA219 ina219;
WiFiClient espClient;
PubSubClient client(espClient);

float vSolar = 0.0;
float vBat = 0.0;
float iUsb = 0.0;
float pUsb = 0.0;
int batPercentage = 0;

float readAnalogVoltage(int pin) {
  int raw = analogRead(pin);
  float vPin = (raw / 4095.0) * 3.3;
  return vPin * 11.0;
}

void publishMqttTelemetry() {
  String json = "{";
  json += "\"vSolar\":" + String(vSolar, 2) + ",";
  json += "\"vBat\":" + String(vBat, 2) + ",";
  json += "\"iUsb\":" + String(iUsb, 3) + ",";
  json += "\"pUsb\":" + String(pUsb, 2) + ",";
  json += "\"batteryPct\":" + String(batPercentage) + ",";
  json += "\"author\":\"M KHOIRUL FAHMI XI C\"";
  json += "}";

  client.publish(mqtt_topic, json.c_str());
}

void setup() {
  Serial.begin(115200);
  pinMode(PIN_STATUS_LED, OUTPUT);
  analogReadResolution(12);

  ina219.begin();
  WiFi.softAP(ssid, password);
  client.setServer(mqtt_server, 1883);
}

void loop() {
  if (!client.connected()) {
    client.connect("ESP32-SOLAR-FAHMI-X1");
  }
  client.loop();

  vSolar = readAnalogVoltage(PIN_SOLAR_ADC);
  vBat = readAnalogVoltage(PIN_BATTERY_ADC);
  iUsb = ina219.getCurrent_mA() / 1000.0;
  pUsb = ina219.getBusVoltage_V() * iUsb;
  batPercentage = (int)constrain(((vBat - 3.0) / 1.2) * 100.0, 0, 100);

  publishMqttTelemetry();
  delay(1000);
}

● KONDISI SIRKUIT: SEHAT & NORMAL
Gejala Klinis (Symptoms Terukur):
Seluruh tegangan berada pada ambang kerja normal (Solar ~5.5V, Baterai ~3.7V, USB 5.05V stabil). Tidak ada lonjakan panas atau arus bocor.
Respon Modul Proteksi (Hardware Response):
Chip DW01A & MT3608 bekerja normal, MOSFET konduktif penuh, LED status hijau/merah menyala sesuai mode.
Analisis Akar Masalah (Root Cause Analysis):
Tidak ada gangguan elektrik. Sambungan solder dan kabel terisolasi baik dengan resistansi kontak minimal (< 0.05 Ω).
Prosedur Perbaikan & Pencegahan (Troubleshooting Steps):
Lakukan inspeksi visual berkala dan pastikan baut terminal serta holder baterai terpasang rapat dan tidak korosi.
